cmake_minimum_required(VERSION 3.16)
project(stb_wuffs LANGUAGES C CXX)
include(CTest)
option(STB_WUFFS_ASAN "Build an AddressSanitizer regression executable" OFF)
add_library(stb_wuffs INTERFACE)
target_include_directories(stb_wuffs INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(stb_wuffs INTERFACE c_std_99)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  foreach(target IN ITEMS adapter reference adapter_cpp)
    if(target STREQUAL "adapter_cpp")
      add_library(${target} SHARED tests/implementation.cpp)
      target_compile_features(${target} PRIVATE cxx_std_11)
      set_target_properties(${target} PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON)
    else()
      add_library(${target} SHARED tests/implementation.c)
      target_compile_features(${target} PRIVATE c_std_11)
    endif()
    if(target STREQUAL "reference")
      target_compile_definitions(${target} PRIVATE TEST_REFERENCE)
    endif()
    set_target_properties(${target} PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
    if(MSVC)
      target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
      target_compile_options(${target} PRIVATE /W4 /wd4505)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra)
      target_link_libraries(${target} PRIVATE m)
    endif()
  endforeach()
  foreach(option IN ITEMS STBI_NO_STDIO STBI_NO_LINEAR STBI_NO_HDR STBI_NO_PNG
      STBI_NO_GIF STBI_NO_JPEG STBI_NO_THREAD_LOCALS STBI_NO_FAILURE_STRINGS
      STBI_FAILURE_USERMSG STBI_NO_SIMD STBI_WINDOWS_UTF8
      STBI_ONLY_PNG STBI_ONLY_JPEG STBI_ONLY_GIF STBI_ONLY_BMP STBI_ONLY_TGA
      STBI_ONLY_PNM STBI_ONLY_PSD STBI_ONLY_HDR STBI_ONLY_PIC STBI_WUFFS_DISABLE STBI_WUFFS_EXACT_JPEG)
    add_executable(compile_${option} tests/compile.c)
    target_compile_features(compile_${option} PRIVATE c_std_11)
    target_compile_definitions(compile_${option} PRIVATE ${option} _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
      target_compile_options(compile_${option} PRIVATE /W4 /wd4505)
    else()
      target_compile_options(compile_${option} PRIVATE -Wall -Wextra -Wno-unused-function)
      target_link_libraries(compile_${option} PRIVATE m)
    endif()
    add_test(NAME compile_${option} COMMAND compile_${option})
  endforeach()
  add_executable(static_link tests/static_a.c tests/static_b.c)
  target_compile_features(static_link PRIVATE c_std_11)
  target_compile_definitions(static_link PRIVATE _CRT_SECURE_NO_WARNINGS)
  if(NOT MSVC)
    target_link_libraries(static_link PRIVATE m)
  endif()
  add_test(NAME static_link COMMAND static_link)
  add_test(NAME differential COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/differential.py"
    "$<TARGET_FILE:adapter>" "$<TARGET_FILE:reference>" "$<TARGET_FILE:adapter_cpp>"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME reproducible COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/amalgamate.py" --check)
  add_test(NAME edge_cases COMMAND ${Python3_EXECUTABLE}
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/edge_cases.py" "$<TARGET_FILE:adapter>" "$<TARGET_FILE:reference>")
  if(STB_WUFFS_ASAN)
    add_executable(sanitizer tests/sanitizer.c)
    target_compile_features(sanitizer PRIVATE c_std_11)
    if(MSVC)
      target_compile_definitions(sanitizer PRIVATE _CRT_SECURE_NO_WARNINGS)
      target_compile_options(sanitizer PRIVATE /fsanitize=address /Zi)
      target_link_options(sanitizer PRIVATE /INCREMENTAL:NO /DEBUG)
    else()
      target_compile_options(sanitizer PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
      target_link_options(sanitizer PRIVATE -fsanitize=address,undefined)
      target_link_libraries(sanitizer PRIVATE m)
    endif()
    add_test(NAME sanitizer COMMAND ${Python3_EXECUTABLE}
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_sanitizer.py" "$<TARGET_FILE:sanitizer>")
  endif()
endif()
